      Довідка з формату 
      scenes.<id>:
  title, text, video, image, kenburns, duration, color
  voice             закадровий голос (аудіофайл)
  music             фонова музика: "файл.mp3" або false (стоп)
  pos {x,y}         місце на схемі редактора
  hotspots[]        {from,to,x,y,w,h, label, hidden, if, set, add, goto}
  events[]          {at, type:"choice"|"qte", ...}
